# sduino - Core Arduino para STM8S (SDCC)
# ----------------------------------------
# Basado en el proyecto sduino de Michael Mayer: https://github.com/tenbaht/sduino
# Adaptado al Arduino IDE 2.x (arduino-cli).
#
# Especificacion: https://arduino.github.io/arduino-cli/latest/platform-specification/

name=STM8S (sduino)
version=0.6.0

# Herramientas
# ------------
# SDCC (compilador): {runtime.tools.sdcc.path}
#   - Gestor de placas: herramienta "sdcc" del paquete.
#   - Instalacion manual: hardware/sduino/tools/sdcc/<version>/ (ver README).
# Scripts auxiliares, stm8flash y stm8gal: van dentro de la plataforma.
sduino.tools.path={runtime.platform.path}/tools

# Los scripts .sh se ejecutan con busybox en Windows
compiler.wrapper.path={sduino.tools.path}/wrapper
compiler.wrapper.path.windows={sduino.tools.path}/win/busybox.exe" ash "{sduino.tools.path}/wrapper

# Compilador
# ----------
compiler.warning_flags=--less-pedantic
compiler.warning_flags.none=--less-pedantic
compiler.warning_flags.default=--less-pedantic
compiler.warning_flags.more=
compiler.warning_flags.all=-Wp-Wall

compiler.path={runtime.tools.sdcc.path}/bin/

# Optimizacion (menu de la placa): por defecto, tamano de codigo
build.opt_flags=--opt-code-size

compiler.c.cmd=sdcc
compiler.c.wrapper=sdcc.sh
compiler.common.flags=--std-sdcc11 -Ddouble=float -DUSE_STDINT -D__PROG_TYPES_COMPAT__
compiler.c.flags=-c -MMD {compiler.common.flags} {build.opt_flags} {compiler.warning_flags}
compiler.cpp.cmd=sdcc
compiler.cpp.wrapper=sdcc.sh
compiler.cpp.flags={compiler.c.flags}
compiler.c.elf.cmd=sdcc
compiler.c.elf.wrapper=sdcc-link.sh
compiler.c.elf.flags=--code-size {upload.maximum_size} --iram-size {upload.maximum_data_size}
compiler.ar.cmd=sdar
compiler.ar.wrapper=sdar.sh
compiler.ar.flags=rcs

compiler.syslibs.stdlib.path={runtime.tools.sdcc.path}/share/sdcc
compiler.syslibs.stdlib.path.windows={runtime.tools.sdcc.path}
compiler.syslibs.spl.path={runtime.platform.path}/STM8S_StdPeriph_Driver
compiler.systemincludes="-I{compiler.syslibs.spl.path}/inc" "-I{compiler.syslibs.stdlib.path}/include"

compiler.c.extra_flags=
compiler.cpp.extra_flags=
compiler.c.elf.extra_flags=
compiler.ar.extra_flags=
build.extra_flags=

build.defines=-mstm8 -D{build.mcu} -DF_CPU={build.f_cpu} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_STM8

# Recetas de compilacion
# ----------------------
recipe.c.o.pattern="{compiler.wrapper.path}/{compiler.c.wrapper}" "{compiler.path}{compiler.c.cmd}" "{source_file}" "{object_file}" c {compiler.c.flags} {build.defines} {compiler.c.extra_flags} {build.extra_flags} {includes} {compiler.systemincludes}

recipe.cpp.o.pattern="{compiler.wrapper.path}/{compiler.cpp.wrapper}" "{compiler.path}{compiler.cpp.cmd}" "{source_file}" "{object_file}" cpp {compiler.cpp.flags} {build.defines} {compiler.cpp.extra_flags} {build.extra_flags} {includes} {compiler.systemincludes}

recipe.ar.pattern="{compiler.wrapper.path}/{compiler.ar.wrapper}" "{compiler.path}{compiler.ar.cmd}" "{archive_file_path}" "{object_file}" ar {compiler.ar.flags} {compiler.ar.extra_flags}

# Enlace: genera {build.project_name}.ihx/.map y lo copia a .hex
recipe.c.combine.pattern="{compiler.wrapper.path}/{compiler.c.elf.wrapper}" "{compiler.path}{compiler.c.elf.cmd}" "{build.path}/{build.project_name}" --nostdlib "-L{compiler.syslibs.spl.path}/lib" "-L{build.path}" "-L{compiler.syslibs.stdlib.path}/lib/stm8" {compiler.c.elf.flags} -mstm8 {compiler.c.elf.extra_flags} {object_files} "{build.path}/{archive_file}" -l{build.mcu} -lstm8

recipe.output.tmp_file={build.project_name}.hex
recipe.output.save_file={build.project_name}.{build.variant}.hex

# Tamano: calculado del .map
recipe.size.pattern="{compiler.wrapper.path}/sdsize.sh" "{build.path}/{build.project_name}.map"
recipe.size.regex=^FLASH\s+([0-9]+)
recipe.size.regex.data=^RAM\s+([0-9]+)

# Preprocesado (deteccion de librerias y prototipos de funciones)
preproc.includes.flags=-M -MG -MP
recipe.preproc.includes="{compiler.wrapper.path}/{compiler.cpp.wrapper}" "{compiler.path}{compiler.cpp.cmd}" "{source_file}" "{preprocessed_file_path}" pre {compiler.common.flags} {preproc.includes.flags} {build.defines} {compiler.cpp.extra_flags} {build.extra_flags} {includes} {compiler.systemincludes}
# Sin -c ni -MMD: con ellos SDCC escribe la lista de dependencias en vez del
# codigo preprocesado y el IDE no puede generar los prototipos de funciones.
preproc.macros.flags=-E -DSDUINO_PREPROC
recipe.preproc.macros="{compiler.wrapper.path}/{compiler.cpp.wrapper}" "{compiler.path}{compiler.cpp.cmd}" "{source_file}" "{preprocessed_file_path}" pre {compiler.common.flags} {preproc.macros.flags} {build.defines} {compiler.cpp.extra_flags} {build.extra_flags} {includes} {compiler.systemincludes}


# Carga con ST-Link (stm8flash)
# -----------------------------
tools.stm8flash.path={sduino.tools.path}/linux
tools.stm8flash.path.windows={sduino.tools.path}/win
tools.stm8flash.path.macosx={sduino.tools.path}/macosx
tools.stm8flash.cmd=stm8flash
tools.stm8flash.cmd.windows=stm8flash.exe

tools.stm8flash.upload.params.verbose=
tools.stm8flash.upload.params.quiet=
tools.stm8flash.upload.pattern="{path}/{cmd}" -c {upload.protocol} -p {upload.mcu} -s flash -w "{build.path}/{build.project_name}.hex"

tools.stm8flash.program.params.verbose=
tools.stm8flash.program.params.quiet=
tools.stm8flash.program.pattern="{path}/{cmd}" -c {program.protocol} -p {upload.mcu} -s flash -w "{build.path}/{build.project_name}.hex"

# "Quemar bootloader": desbloquea un micro protegido contra lectura
# (borra toda la memoria) y restaura los option bytes de fabrica
tools.stm8flash.erase.params.verbose=
tools.stm8flash.erase.params.quiet=
tools.stm8flash.erase.pattern="{path}/{cmd}" -c {program.protocol} -p {upload.mcu} -u
tools.stm8flash.bootloader.params.verbose=
tools.stm8flash.bootloader.params.quiet=
tools.stm8flash.bootloader.pattern="{path}/{cmd}" -c {program.protocol} -p {upload.mcu} -u


# Carga por el puerto serie (bootloader ROM de ST)
# ------------------------------------------------
tools.stm8gal.path={sduino.tools.path}/linux
tools.stm8gal.path.windows={sduino.tools.path}/win
tools.stm8gal.path.macosx={sduino.tools.path}/macosx
tools.stm8gal.cmd=stm8gal
tools.stm8gal.cmd.windows=stm8gal.exe

tools.stm8gal.upload.params.quiet=-v 0
tools.stm8gal.upload.params.verbose=-v 2
tools.stm8gal.upload.verify=
tools.stm8gal.upload.params.noverify=-V
tools.stm8gal.upload.pattern="{path}/{cmd}" -p "{serial.port}" {upload.reset} -B {upload.verbose} {upload.verify} -w "{build.path}/{build.project_name}.hex"

# "Subir usando programador" con el programador "Serial STM8 bootloader"
tools.stm8gal.program.params.quiet=-v 0
tools.stm8gal.program.params.verbose=-v 2
tools.stm8gal.program.verify=
tools.stm8gal.program.params.noverify=-V
tools.stm8gal.program.pattern="{path}/{cmd}" -p "{serial.port}" {upload.reset} -B {program.verbose} {program.verify} -w "{build.path}/{build.project_name}.hex"
